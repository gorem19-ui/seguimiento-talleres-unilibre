const TZ = 'America/Bogota';
const FESTIVOS = { '2026-11-16': 'Independencia de Cartagena', '2026-12-08': 'Inmaculada Concepción' };
const PERIODO_INICIO = '2026-11-09';
const PERIODO_FIN = '2026-12-15';
const DIAS = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado'];
const NOVEDADES = [
  'Cambio de horario acordado con la coordinación',
  'Reposición de una sesión',
  'Actividad en otro escenario o fuera del campus (evento, torneo, salida)',
  'Retraso por fuerza mayor (transporte, clima, salud)',
  'Falla del GPS o de la señal del celular',
  'Otra'
];
const ENC_ASIS = ['registro', 'fecha', 'hora_registro', 'id', 'taller', 'profesor', 'sesion', 'tipo', 'hora_programada',
  'lat', 'lng', 'precision_m', 'origen_ubicacion', 'distancia_m', 'dentro_radio', 'foto_url', 'hora_foto', 'estado',
  'observacion', 'fuera_horario', 'fuera_rango', 'novedad', 'novedad_detalle'];

// ---------------------------------------------------------------- web
function doGet() {
  const url = cfg_().PAGINA_URL || '';
  return HtmlService.createHtmlOutput('<p style="font-family:sans-serif">La página de seguimiento está en <a href="' + url + '" target="_top">' + url + '</a></p>')
    .setTitle('Seguimiento semanal de talleres');
}

function doPost(e) {
  let out;
  try {
    const b = JSON.parse(e.postData.contents);
    const F = { login: login, hoy: hoy, miSemana: miSemana, evaluar: evaluar, registrar: registrar, subsanar: subsanar, panel: panel };
    if (!F[b.fn]) throw new Error('Función no válida.');
    out = { ok: true, data: F[b.fn].apply(null, b.args || []) };
  } catch (err) {
    out = { ok: false, error: err && err.message ? err.message : String(err) };
  }
  return ContentService.createTextOutput(JSON.stringify(out)).setMimeType(ContentService.MimeType.JSON);
}

// ---------------------------------------------------------------- utilidades
function ss_() { return SpreadsheetApp.getActive(); }
function hoja_(n) { const h = ss_().getSheetByName(n); if (!h) throw new Error('Falta la hoja ' + n + '.'); return h; }
const MEMO_ = {};
function olvidar_(n) { delete MEMO_[n]; }
function filas_(n) {
  if (MEMO_[n]) return MEMO_[n];
  const v = hoja_(n).getDataRange().getDisplayValues();
  const enc = v.shift();
  MEMO_[n] = v.filter(r => r.join('') !== '').map(r => { const o = {}; enc.forEach((k, i) => o[k] = r[i]); return o; });
  return MEMO_[n];
}
function cfg_() { const o = {}; filas_('Config').forEach(r => o[r.clave] = r.valor); return o; }
function hoyISO_() { return Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd'); }
function ahoraHM_() { return Utilities.formatDate(new Date(), TZ, 'HH:mm'); }
function d_(s) { const p = s.split('-').map(Number); return new Date(Date.UTC(p[0], p[1] - 1, p[2])); }
function iso_(d) { return d.toISOString().slice(0, 10); }
function sumD_(s, n) { const d = d_(s); d.setUTCDate(d.getUTCDate() + n); return iso_(d); }
function min_(t) { const p = String(t).split(':').map(Number); return p[0] * 60 + (p[1] || 0); }
function hm_(m) { return String(Math.floor(m / 60)).padStart(2, '0') + ':' + String(m % 60).padStart(2, '0'); }
function lunes_(s) { const w = d_(s).getUTCDay(); return sumD_(s, w === 0 ? -6 : 1 - w); }
function semanas_() { const o = []; for (let s = PERIODO_INICIO; s <= PERIODO_FIN; s = sumD_(s, 7)) o.push(s); return o; }
function semanaActual_() { const h = hoyISO_(); let a = semanas_()[0]; semanas_().forEach(s => { if (s <= h) a = s; }); return a; }
function prof_(id) { const p = filas_('Profesores').find(r => r.id === id); if (!p) throw new Error('Taller no encontrado.'); return p; }
function token_(tok) {
  const id = tok && CacheService.getScriptCache().get('t_' + tok);
  if (!id) throw new Error('Su sesión venció. Ingrese de nuevo con su cédula.');
  return id;
}
function distancia_(a, b, c, d) {
  const R = 6371000, r = x => x * Math.PI / 180;
  const h = Math.sin(r(c - a) / 2) ** 2 + Math.cos(r(a)) * Math.cos(r(c)) * Math.sin(r(d - b) / 2) ** 2;
  return Math.round(2 * R * Math.asin(Math.sqrt(h)));
}

// Programación de la semana: la fijada por coordinación para esa semana (tipo "base" en
// la hoja Concertacion), o el horario de la hoja Horarios.
function programa_(id, lunes) {
  try {
    const pb = filas_('Concertacion').filter(r => r.id === id && r.semana === lunes && r.tipo === 'base');
    if (pb.length) return JSON.parse(pb[pb.length - 1].sesiones_nuevas).map(x => Object.assign({}, x, { festivo: FESTIVOS[x.fecha] || '' }));
  } catch (e) { }
  return filas_('Horarios').filter(r => r.id === id).map(r => {
    const dia = Number(r.dia);
    const fecha = sumD_(lunes, dia === 0 ? 6 : dia - 1);
    return { fecha: fecha, inicio: r.inicio, fin: r.fin, etiqueta: r.etiqueta || '', festivo: FESTIVOS[fecha] || '' };
  }).filter(x => x.fecha <= PERIODO_FIN).sort((a, b) => (a.fecha + hm_(min_(a.inicio))).localeCompare(b.fecha + hm_(min_(b.inicio))));
}
function asistencia_() {
  const h = hoja_('Asistencia');
  const enc = h.getRange(1, 1, 1, Math.max(h.getLastColumn(), 1)).getValues()[0];
  if (enc.length < ENC_ASIS.length || enc[ENC_ASIS.length - 1] !== ENC_ASIS[ENC_ASIS.length - 1]) {
    h.getRange(1, 1, 1, ENC_ASIS.length).setValues([ENC_ASIS]).setFontWeight('bold');
    olvidar_('Asistencia');
  }
  return filas_('Asistencia');
}

// ---------------------------------------------------------------- API profesor
function login(cedula) {
  const cc = String(cedula || '').replace(/\D/g, '');
  if (cc.length < 5) throw new Error('Escriba su número de cédula sin puntos.');
  const p = filas_('Profesores').find(r => String(r.cedula).replace(/\D/g, '') === cc);
  if (!p) throw new Error('La cédula no está registrada en el programa. Comuníquese con la coordinación de RECREAVALLE.');
  const tok = Utilities.getUuid();
  CacheService.getScriptCache().put('t_' + tok, p.id, 21600);
  return { token: tok, prof: { id: p.id, area: p.area, taller: p.taller, profesor: p.profesor, lugar: p.lugar }, novedades: NOVEDADES };
}

function hoy(tok) {
  const id = token_(tok);
  revisar_([id]);
  const f = hoyISO_();
  const ses = programa_(id, lunes_(f)).filter(x => x.fecha === f);
  const regs = asistencia_().filter(r => r.id === id && r.fecha === f);
  const abierta = abierta_(regs);
  return {
    hoy: f, ahora: ahoraHM_(), festivo: FESTIVOS[f] || '',
    sesiones: ses.map(x => ({ inicio: x.inicio, fin: x.fin, etiqueta: x.etiqueta })),
    registros: regs.map(r => ({ hora: r.hora_registro, tipo: r.tipo, sesion: r.sesion, estado: r.estado, novedad: r.novedad })),
    siguiente: abierta ? 'salida' : 'llegada',
    pendientes: filas_('Pendientes').filter(r => r.id === id && r.estado === 'Pendiente').map(r => ({ clave: r.clave, fecha: r.fecha, sesion: r.sesion, falta: r.falta }))
  };
}

function miSemana(tok) {
  const id = token_(tok);
  revisar_([id]);
  const l = semanaActual_(), f = hoyISO_(), ahora = min_(ahoraHM_());
  const asis = asistencia_().filter(r => r.id === id && r.fecha >= l && r.fecha <= sumD_(l, 6));
  const est = r => r ? { hora: r.hora_registro, novedad: r.fuera_horario === 'Sí' || r.fuera_rango === 'Sí', extemporaneo: /extemporáneo/.test(r.estado || ''), estado: r.estado } : null;
  const sesiones = programa_(id, l).map(x => {
    const k = x.inicio + '-' + x.fin;
    const ll = asis.find(r => r.fecha === x.fecha && r.sesion === k && r.tipo === 'llegada');
    const sa = asis.find(r => r.fecha === x.fecha && r.sesion === k && r.tipo === 'salida');
    const vencida = x.fecha < f || (x.fecha === f && ahora > min_(x.fin));
    return Object.assign({}, x, { llegada: est(ll), salida: est(sa), vencida: vencida, enCurso: x.fecha === f && ahora >= min_(x.inicio) && ahora <= min_(x.fin) });
  });
  return { semana: l, sesiones: sesiones, festivos: FESTIVOS };
}

function abierta_(regs) {
  let ab = null;
  regs.forEach(r => { if (r.tipo === 'llegada') ab = r; else if (r.tipo === 'salida') ab = null; });
  return ab;
}

// Calcula si el registro queda fuera de horario o fuera de rango.
function calcular_(id, tipo, lat, lng) {
  const c = cfg_(), p = prof_(id);
  const f = hoyISO_(), ahoraS = ahoraHM_(), ahora = min_(ahoraS);
  const tol = Number(c.TOLERANCIA_MIN || 10);
  const ses = programa_(id, lunes_(f)).filter(x => x.fecha === f && !x.festivo);
  const regs = asistencia_().filter(r => r.id === id && r.fecha === f);
  const ab = abierta_(regs);
  if (tipo !== 'llegada' && tipo !== 'salida') throw new Error('Tipo de registro no válido.');
  if (tipo === 'llegada' && ab) throw new Error('Ya tiene una llegada abierta a las ' + ab.hora_registro + '. Registre primero la salida.');
  if (tipo === 'salida' && !ab) throw new Error('No hay una llegada registrada hoy. Registre primero la llegada.');

  let sesion = '', programada = '', horario = '';
  if (tipo === 'llegada') {
    // sesión más cercana a la hora actual
    let mejor = null, dmin = 1e9;
    ses.forEach(x => { const d = Math.abs(ahora - min_(x.inicio)); if (d < dmin && ahora <= min_(x.fin)) { dmin = d; mejor = x; } });
    if (!mejor) horario = ses.length ? 'Sin sesión programada a esta hora' : (FESTIVOS[f] ? 'Día festivo, sin sesión' : 'Sin sesión programada hoy');
    else {
      sesion = mejor.inicio + '-' + mejor.fin; programada = mejor.inicio;
      const ini = min_(mejor.inicio);
      if (ahora > ini + tol) horario = 'Llegada tarde (' + (ahora - ini) + ' min)';
      else if (ahora < ini - 30) horario = 'Llegada anticipada (' + (ini - ahora) + ' min antes)';
    }
  } else {
    sesion = ab.sesion;
    if (sesion) {
      const fin = min_(sesion.split('-')[1]); programada = sesion.split('-')[1];
      if (ahora < fin - tol) horario = 'Salida anticipada (' + (fin - ahora) + ' min antes)';
      else if (ahora > fin + 30) horario = 'Salida tardía (' + (ahora - fin) + ' min después)';
    } else horario = 'Salida de una llegada sin sesión programada';
  }

  let sLat = Number(c.SEDE_LAT), sLng = Number(c.SEDE_LNG);
  if (p.lat && p.lng) { sLat = Number(p.lat); sLng = Number(p.lng); }
  const radio = Number(c.RADIO_M || 300);
  const tiene = lat !== null && lat !== undefined && lat !== '' && !isNaN(Number(lat));
  let dist = '', rango = '';
  if (!tiene) rango = 'Sin ubicación';
  else if (sLat && sLng) { dist = distancia_(Number(lat), Number(lng), sLat, sLng); if (dist > radio) rango = 'Fuera del rango de la sede (' + dist + ' m)'; }
  return { fecha: f, hora: ahoraS, sesion: sesion, programada: programada, fueraHorario: horario, fueraRango: rango, dist: dist, tiene: tiene, prof: p, cfg: c };
}

function evaluar(tok, d) {
  const id = token_(tok);
  const r = calcular_(id, d.tipo, d.lat, d.lng);
  return { hora: r.hora, sesion: r.sesion, fueraHorario: r.fueraHorario, fueraRango: r.fueraRango, requiereNovedad: !!(r.fueraHorario || r.fueraRango) };
}

function registrar(tok, d) {
  const id = token_(tok);
  if (!d.foto) throw new Error('Tome la foto de evidencia.');
  const r = calcular_(id, d.tipo, d.lat, d.lng);
  const nov = String(d.novedad || '').trim(), det = String(d.detalle || '').trim().slice(0, 500);
  if (r.fueraHorario || r.fueraRango) {
    if (NOVEDADES.indexOf(nov) < 0) throw new Error('Seleccione el motivo de la novedad.');
    if (nov === 'Otra' && det.length < 5) throw new Error('Explique el motivo de la novedad.');
  }
  let fotoHora = String(d.fotoHora || '');
  const est = [];
  if (r.fueraHorario) est.push(r.fueraHorario); else est.push(d.tipo === 'llegada' ? 'A tiempo' : 'Completa');
  if (r.fueraRango) est.push(r.fueraRango);
  if (fotoHora && (fotoHora.slice(0, 10) !== r.fecha || Math.abs(min_(fotoHora.slice(11)) - min_(r.hora)) > 30)) est.push('Foto no reciente (' + fotoHora + ')');
  const carpeta = DriveApp.getFolderById(r.cfg.CARPETA_FOTOS_ID);
  const f = carpeta.createFile(Utilities.newBlob(Utilities.base64Decode(d.foto), 'image/jpeg', r.fecha + '_' + id + '_' + d.tipo + '_' + r.hora.replace(':', '') + '.jpg'));
  const fila = [new Date(), r.fecha, r.hora, id, r.prof.taller, r.prof.profesor, r.sesion, d.tipo, r.programada,
    r.tiene ? Number(d.lat) : '', r.tiene ? Number(d.lng) : '', d.precision || '', d.origen || '', r.dist,
    r.fueraRango ? (r.tiene ? 'No' : 'Sin ubicación') : (r.dist === '' ? 'Sin referencia' : 'Sí'),
    f.getUrl(), fotoHora, est.join(' · '), String(d.obs || '').slice(0, 300),
    r.fueraHorario ? 'Sí' : 'No', r.fueraRango ? 'Sí' : 'No', (r.fueraHorario || r.fueraRango) ? nov : '', (r.fueraHorario || r.fueraRango) ? det : ''];
  const h = hoja_('Asistencia');
  h.getRange(h.getLastRow() + 1, 1, 1, fila.length).setNumberFormat('@').setValues([fila.map(x => x instanceof Date ? Utilities.formatDate(x, TZ, 'yyyy-MM-dd HH:mm:ss') : String(x))]);
  olvidar_('Asistencia');
  return { ok: true, hora: r.hora, estado: est.join(' · ') };
}

// ---------------------------------------------------------------- API panel
function panel(clave, lunes) {
  const c = cfg_();
  if (!clave || clave !== c.ADMIN_CLAVE) throw new Error('Clave incorrecta.');
  const sems = semanas_(); const f = hoyISO_(); const ahora = min_(ahoraHM_());
  if (!lunes) lunes = semanaActual_();
  revisar_(null);
  const fin = sumD_(lunes, 6);
  const asis = asistencia_().filter(r => r.fecha >= lunes && r.fecha <= fin);
  const talleres = filas_('Profesores').map(p => {
    const prog = programa_(p.id, lunes).filter(x => !x.festivo);
    const mios = asis.filter(r => r.id === p.id);
    const sesiones = prog.map(x => {
      const k = x.inicio + '-' + x.fin;
      const ll = mios.find(r => r.fecha === x.fecha && r.sesion === k && r.tipo === 'llegada');
      const sa = mios.find(r => r.fecha === x.fecha && r.sesion === k && r.tipo === 'salida');
      return { fecha: x.fecha, sesion: k, etiqueta: x.etiqueta, vencida: x.fecha < f || (x.fecha === f && min_(x.fin) < ahora), llegada: ll ? res_(ll) : null, salida: sa ? res_(sa) : null };
    });
    return { id: p.id, area: p.area, taller: p.taller, profesor: p.profesor, telefono: p.telefono, sesiones: sesiones, registros: mios.map(res_) };
  });
  const pend = filas_('Pendientes').map(r => ({ fecha: r.fecha, taller: r.taller, profesor: r.profesor, id: r.id, sesion: r.sesion, falta: r.falta, detectado: r.detectado, estado: r.estado, subsanado: r.subsanado_el, novedad: r.novedad, detalle: r.detalle, foto: r.foto_url }));
  return { semana: lunes, semanas: sems, festivos: FESTIVOS, talleres: talleres, pendientes: pend, url: c.PAGINA_URL || '', correos: c.CORREOS_NOVEDADES || '' };
}
function res_(r) {
  return { fecha: r.fecha, hora: r.hora_registro, tipo: r.tipo, sesion: r.sesion, estado: r.estado, foto: r.foto_url, dist: r.distancia_m,
    fueraHorario: r.fuera_horario === 'Sí', fueraRango: r.fuera_rango === 'Sí', novedad: r.novedad || '', detalle: r.novedad_detalle || '', obs: r.observacion || '' };
}

// ---------------------------------------------------------------- pendientes
const ENC_PEND = ['clave', 'fecha', 'id', 'taller', 'profesor', 'sesion', 'falta', 'detectado', 'estado', 'subsanado_el', 'novedad', 'detalle', 'foto_url'];
function hojaPend_() {
  let h = ss_().getSheetByName('Pendientes');
  if (!h) { h = ss_().insertSheet('Pendientes'); h.getRange(1, 1, 1, ENC_PEND.length).setValues([ENC_PEND]).setFontWeight('bold'); h.setFrozenRows(1); h.getRange('A:M').setNumberFormat('@'); olvidar_('Pendientes'); }
  return h;
}

// Anota en "Pendientes" las sesiones vencidas sin llegada o sin salida, y cierra las que ya tienen registro.
function revisar_(ids) {
  const h = hojaPend_();
  const c = cfg_();
  const desde = c.FECHA_INICIO_CONTROL || '2026-11-12';
  const f = hoyISO_(), ahora = min_(ahoraHM_());
  const asis = asistencia_();
  const pend = filas_('Pendientes');
  const existentes = {}; pend.forEach((r, i) => existentes[r.clave] = { fila: i + 2, r: r });
  const nuevas = [], cerrar = [];
  const profs = filas_('Profesores').filter(p => !ids || ids.indexOf(p.id) >= 0);
  const marca = Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd HH:mm');
  profs.forEach(p => {
    for (let l = lunes_(desde); l <= lunes_(f); l = sumD_(l, 7)) {
      programa_(p.id, l).forEach(x => {
        if (x.festivo || x.fecha < desde || x.fecha > f) return;
        const fin = min_(x.fin), k = x.inicio + '-' + x.fin;
        const regs = asis.filter(r => r.id === p.id && r.fecha === x.fecha && r.sesion === k);
        const ll = regs.some(r => r.tipo === 'llegada'), sa = regs.some(r => r.tipo === 'salida');
        const faltas = [];
        if (x.fecha < f || ahora > fin) { if (!ll) faltas.push('llegada'); }
        if (x.fecha < f || ahora > fin + 60) { if (ll && !sa) faltas.push('salida'); }
        ['llegada', 'salida'].forEach(t => {
          const clave = x.fecha + '|' + p.id + '|' + k + '|' + t;
          const ex = existentes[clave];
          const tiene = t === 'llegada' ? ll : sa;
          if (faltas.indexOf(t) >= 0 && !ex) nuevas.push([clave, x.fecha, p.id, p.taller, p.profesor, k, t, marca, 'Pendiente', '', '', '', '']);
          if (ex && ex.r.estado === 'Pendiente' && tiene) cerrar.push(ex.fila);
        });
      });
    }
  });
  if (nuevas.length) h.getRange(h.getLastRow() + 1, 1, nuevas.length, ENC_PEND.length).setNumberFormat('@').setValues(nuevas);
  cerrar.forEach(fi => h.getRange(fi, 9, 1, 2).setValues([['Subsanado', marca]]));
  if (nuevas.length || cerrar.length) olvidar_('Pendientes');
}

// El profesor sube la evidencia de una sesión pendiente, con el motivo.
function subsanar(tok, d) {
  const id = token_(tok);
  const p = prof_(id), c = cfg_();
  const pend = filas_('Pendientes');
  const i = pend.findIndex(r => r.clave === d.clave && r.id === id);
  if (i < 0) throw new Error('Pendiente no encontrado.');
  const r = pend[i];
  if (r.estado !== 'Pendiente') throw new Error('Este pendiente ya fue subsanado.');
  if (!d.foto) throw new Error('Adjunte la foto de evidencia de la sesión.');
  const nov = String(d.novedad || '').trim(), det = String(d.detalle || '').trim().slice(0, 500);
  if (NOVEDADES.indexOf(nov) < 0) throw new Error('Seleccione el motivo.');
  if (det.length < 5) throw new Error('Explique por qué no registró a tiempo.');
  const ahoraS = ahoraHM_(), hoyS = hoyISO_(), marca = hoyS + ' ' + ahoraS;
  const carpeta = DriveApp.getFolderById(c.CARPETA_FOTOS_ID);
  const f = carpeta.createFile(Utilities.newBlob(Utilities.base64Decode(d.foto), 'image/jpeg', r.fecha + '_' + id + '_' + r.falta + '_extemporaneo.jpg'));
  let sLat = Number(c.SEDE_LAT), sLng = Number(c.SEDE_LNG);
  if (p.lat && p.lng) { sLat = Number(p.lat); sLng = Number(p.lng); }
  const tiene = d.lat !== null && d.lat !== undefined && d.lat !== '' && !isNaN(Number(d.lat));
  let dist = '', rango = '';
  if (!tiene) rango = 'Sin ubicación';
  else if (sLat && sLng) { dist = distancia_(Number(d.lat), Number(d.lng), sLat, sLng); if (dist > Number(c.RADIO_M || 300)) rango = 'Fuera del rango de la sede (' + dist + ' m)'; }
  const est = ['Registro extemporáneo (subido el ' + marca + ')'];
  if (rango) est.push(rango + ' al subir');
  const fila = [marca, r.fecha, ahoraS, id, p.taller, p.profesor, r.sesion, r.falta, r.falta === 'llegada' ? r.sesion.split('-')[0] : r.sesion.split('-')[1],
    tiene ? Number(d.lat) : '', tiene ? Number(d.lng) : '', d.precision || '', d.origen || '', dist, rango ? (tiene ? 'No' : 'Sin ubicación') : (dist === '' ? 'Sin referencia' : 'Sí'),
    f.getUrl(), String(d.fotoHora || ''), est.join(' · '), '', 'Sí', rango ? 'Sí' : 'No', nov, det];
  const filas = [fila];
  const finS = r.sesion.split('-')[1];
  if (r.falta === 'llegada' && (r.fecha < hoyS || min_(ahoraS) > min_(finS) + 60)) {
    const sal = fila.slice(); sal[7] = 'salida'; sal[8] = finS; filas.push(sal);  // la sesión completa queda subsanada
  }
  const ha = hoja_('Asistencia');
  ha.getRange(ha.getLastRow() + 1, 1, filas.length, fila.length).setNumberFormat('@').setValues(filas.map(x => x.map(String)));
  olvidar_('Asistencia');
  hojaPend_().getRange(i + 2, 9, 1, 5).setValues([['Subsanado (extemporáneo)', marca, nov, det, f.getUrl()]]);
  olvidar_('Pendientes');
  return { ok: true, estado: est.join(' · ') };
}
